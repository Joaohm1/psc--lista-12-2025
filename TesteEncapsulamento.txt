package Aplicacao;

public class TesteEncapsulamento {

	public static void main(String[] args) {
System.out.println("---  Teste de Encapsulamento da Conta Bancária ---");
        
        // 1. Cria uma nova instância de ContaBancaria
        // Usando o construtor que internamente chama os Setters
        ContaBancaria conta = new ContaBancaria("12345-6", "Alice Silva");
        
        System.out.println("\n** Dados Iniciais da Conta (via Getters) **");
        System.out.println("Número da Conta: " + conta.getNumeroConta());
        System.out.println("Titular: " + conta.getTitular());
        System.out.println("Saldo: R$" + conta.getSaldo());
        
        // --- Ponto 2: Tentativa de Acesso Direto (Causará Erro de Compilação) ---
        // Descomente as linhas abaixo para ver o erro em seu ambiente de desenvolvimento:
        // conta.saldo = 500.0;    // ERRO: O campo 'saldo' não está visível
        // conta.titular = "Bob";  // ERRO: O campo 'titular' não está visível

        // --- Ponto 3: Uso de Setters para modificar e validar ---
        
        System.out.println("\n** Teste de Validação dos Setters **");
        // Tentativa de definir valor inválido (vazio)
        conta.setTitular(""); // Vai imprimir ERRO
        // Tentativa de definir valor válido
        conta.setTitular("Alice Maravilha"); 
        
        System.out.println("Titular Atualizado: " + conta.getTitular());

        // --- Ponto 4: Teste dos Métodos de Transação ---

        System.out.println("\n** Teste de Transações (Depósito) **");
        
        // Depósito com sucesso
        conta.depositar(200.50);
        
        // Tentativa de depósito inválido (valor negativo)
        conta.depositar(-10.00); 

        System.out.println("\n** Teste de Transações (Saque) **");
        
        // Saque com sucesso
        conta.sacar(50.00);
        
        // Tentativa de saque inválido (valor zero ou negativo)
        conta.sacar(0);
        
        // Tentativa de saque maior que o saldo atual
        conta.sacar(5000.00); 

        // --- Ponto 5: Saldo Final ---
        System.out.println("\n** Saldo Final (via Getter) **");
        System.out.printf("Saldo Final da conta : R$", conta.getNumeroConta(), conta.getSaldo());
    }

	}
